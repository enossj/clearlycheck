```js
function letterFor(p) {
  const t = [[97,'A+'],[93,'A'],[90,'A-'],[87,'B+'],[83,'B'],[80,'B-'],[77,'C+'],[73,'C'],[70,'C-'],[67,'D+'],[63,'D'],[60,'D-'],[0,'F']];
  for (const [min, l] of t) if (p >= min) return l;
  return 'F';
}
function computeTestGrade({ mode, total, wrong, earned, possible }) {
  let percent;
  if (mode === 'points') {
    if (possible <= 0) return { percent: 0, letter: 'F' };
    percent = earned / possible * 100;
  } else {
    if (total <= 0) return { percent: 0, letter: 'F' };
    const w = Math.min(Math.max(wrong, 0), total);
    percent = (total - w) / total * 100;
  }
  return { percent, letter: letterFor(percent) };
}
const a = computeTestGrade({ mode: 'questions', total: 20, wrong: 1 });
console.assert(Math.abs(a.percent - 95) < 1e-9 && a.letter === 'A', 'a', a);
const b = computeTestGrade({ mode: 'questions', total: 50, wrong: 7 });
console.assert(Math.abs(b.percent - 86) < 1e-9 && b.letter === 'B', 'b', b);
const c = computeTestGrade({ mode: 'points', earned: 45, possible: 50 });
console.assert(Math.abs(c.percent - 90) < 1e-9 && c.letter === 'A-', 'c', c);
console.log('OK', a, b, c);
```

```js
const FF = { 'Times New Roman': 1.0, 'Arial': 1.0, 'Calibri': 1.05, 'Courier New': 0.75 };
const SF = { 12: 1.0, 11: 1.10 };
const SP = { single: 1.0, '1.5': 0.667, double: 0.5 };
function wordsPerPage({ font, size, spacing }) { return 500 * FF[font] * SF[size] * SP[spacing]; }
function wordsToPages({ words, font, size, spacing }) { return words / wordsPerPage({ font, size, spacing }); }
function pagesToWords({ pages, font, size, spacing }) { return Math.round(pages * wordsPerPage({ font, size, spacing })); }
const p1 = wordsToPages({ words: 1000, font: 'Times New Roman', size: 12, spacing: 'double' });
console.assert(Math.abs(p1 - 4.0) < 1e-9, 'p1', p1);
const p2 = wordsToPages({ words: 500, font: 'Times New Roman', size: 12, spacing: 'single' });
console.assert(Math.abs(p2 - 1.0) < 1e-9, 'p2', p2);
const w1 = pagesToWords({ pages: 3, font: 'Arial', size: 12, spacing: 'double' });
console.assert(w1 === 750, 'w1', w1);
console.log('OK', p1, p2, w1);
```
